evidence = Path(__file__).resolve().parent
root = evidence.parents[3]
source = json_read(evidence / "tested-source-binding.json")
complete = (evidence / "delivery-inspection.json").exists()
inspection = json_read(evidence / ("delivery-inspection.json" if complete else "package-inspection.json"))

    assert current == baseline, path
    constraints.append(dict(file_record(root / path), GitBlob=current))
