using Microsoft.Extensions.Time.Testing;

internal sealed class ObservedTimeProvider : TimeProvider
{
    private readonly FakeTimeProvider _inner = new(new DateTimeOffset(2026, 10, 6, 0, 0, 0, TimeSpan.Zero));
    internal Action<long>? AfterTimestampCaptured { get; set; }
    internal Func<ITimer, ITimer>? DecorateTimer { get; set; }
    internal Action<TimeSpan>? TimerCreated { get; set; }
    public override long TimestampFrequency => _inner.TimestampFrequency;
    public override DateTimeOffset GetUtcNow() => _inner.GetUtcNow();
    public override long GetTimestamp()
    {
        long captured = _inner.GetTimestamp();
        AfterTimestampCaptured?.Invoke(captured);
        return captured;
    }
    public override ITimer CreateTimer(TimerCallback callback, object? state, TimeSpan dueTime, TimeSpan period)
    {
        var timer = _inner.CreateTimer(callback, state, dueTime, period);
        try
        {
            var result = DecorateTimer?.Invoke(timer) ?? timer;
            TimerCreated?.Invoke(dueTime);
            return result;
        }
        catch { timer.Dispose(); throw; }
    }
    internal void Advance(TimeSpan elapsed) => _inner.Advance(elapsed);
}
