using Iris.App.Capture;
using Iris.App.Tests.Capture;
using Iris.Core.Presentation;

// Differential oracle: after the actual owner has completed each new intent, an old restoration
// must preserve its observed visibility. No product version fields or copied transition table.
internal static class VisibilitySequenceChecks
{
    internal static async Task Run(Func<string, Func<Task>, Task> run)
    {
        int count = 0;
        foreach (var sequence in Sequences("", 3).Where(s => s.Length > 0))
        {
            count++;
            await run("VISIBILITY_SEQUENCE_" + sequence, () => PreparationTestScope.Run(async f =>
            {
                var operations = new List<ITranslateWindowOperation>();
                ITranslateWindowOperation? current = null;
                ITranslateWindowOperation Begin()
                {
                    current = f.Mouse.Presenter.TryBeginOperation() ?? throw new InvalidOperationException("Owner admission unexpectedly closed.");
                    operations.Add(current);
                    return current;
                }
                try
                {
                    await f.Mouse.Presenter.TryShowAsync(Begin(), new() { AutoTranslate = false, FocusInput = false });
                    var window = f.Mouse.Presenter.Window!;
                    var native = new Visibility(); f.Preparation.Visibility = native;
                    var lease = await f.Preparation.PrepareAsync(true);
                    bool entered = false, expected = false;
                    native.BeforeShow = _ =>
                    {
                        if (entered) return;
                        entered = true;
                        foreach (var action in sequence)
                        {
                            if (action == 'B') { Begin(); continue; }
                            if (action == 'H')
                            {
                                var hide = f.Mouse.Presenter.HideAsync(); f.Work.Add(hide);
                                Require(hide.IsCompletedSuccessfully, "Owner Hide did not settle within the injected boundary.");
                                current = null;
                                continue;
                            }
                            if (current == null || current.CancellationToken.IsCancellationRequested) Begin();
                            if (action == 'F') f.Mouse.Position.Result = _ => false;
                            try
                            {
                                var show = f.Mouse.Presenter.TryShowAsync(current!, new()
                                    { Text = "synthetic sequence", AutoTranslate = false, FocusInput = false });
                                f.Work.Add(show);
                                Require(show.IsCompletedSuccessfully, "Owner publication did not settle within the injected boundary.");
                                Require(show.Result == (action == 'S'), "Actual native success/failure fixture was not established.");
                            }
                            finally { f.Mouse.Position.Result = null; }
                        }
                        expected = lease.Reason == PreparationRevocation.None || window.IsVisible;
                    };
                    try
                    {
                        await lease.DisposeAsync();
                        Require(entered, "Old restoration boundary was not entered.");
                        bool actual = window.IsVisible;
                        Console.WriteLine($"SEQUENCE {sequence} expectedVisible={expected} actualVisible={actual} reason={lease.Reason}");
                        Require(actual == expected, "Old restoration overwrote the actual owner outcome for sequence " + sequence);
                    }
                    finally { native.BeforeShow = null; f.Mouse.Position.Result = null; }
                }
                finally { foreach (var operation in operations) operation.Dispose(); }
            }));
        }
        Console.WriteLine($"SEQUENCE_COUNT={count}; B=begin S=successful_show H=hide F=failed_native_show");
    }

    private static IEnumerable<string> Sequences(string prefix, int remaining)
    {
        yield return prefix;
        if (remaining == 0) yield break;
        foreach (char item in "BSHF")
            foreach (var sequence in Sequences(prefix + item, remaining - 1)) yield return sequence;
    }
    private static void Require(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
