internal sealed class Store : ISettingsService
{
    public AppSettings Current { get; private set; } = new() { General = new() { LaunchAtStartup = false, Theme = AppTheme.Light } };
    public event EventHandler<SettingsChangedEventArgs>? Changed { add { } remove { } }
    public Task LoadAsync(CancellationToken token = default) => Task.CompletedTask;
    public Task UpdateAsync(Func<AppSettings, AppSettings> mutate, CancellationToken token = default)
    {
        Current = mutate(Current);
        return Task.CompletedTask;
    }
}
internal sealed class AutoStart : IAutoStartService, IDisposable
{
    internal readonly ManualResetEventSlim Release = new();
    internal readonly TaskCompletionSource FreshQueryEntered = new(TaskCreationOptions.RunContinuationsAsynchronously);
    internal int Queries, Applies;
    public bool IsEnabled()
    {
        if (Interlocked.Increment(ref Queries) > 1)
        {
            FreshQueryEntered.TrySetResult();
            Release.Wait();
        }
        return false;
    }
    public void SetEnabled(bool value) { Interlocked.Increment(ref Applies); throw new IOException("synthetic application failure"); }
    public void Dispose() => Release.Dispose();
}
internal sealed class HoldInitialQuery : IUiDispatcher
{
    private readonly WpfUiDispatcher _real = new();
    private readonly TaskCompletionSource _release = new(TaskCreationOptions.RunContinuationsAsynchronously);
    private Func<Task>? _initial;
    private int _first = 1;
    internal bool Released => _release.Task.IsCompleted;
    internal readonly TaskCompletionSource Queued = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public bool CheckAccess() => _real.CheckAccess();
    public Task InvokeAsync(Func<Task> callback, CancellationToken token)
    {
        if (Interlocked.Exchange(ref _first, 0) == 1)
        {
            _initial = callback; Queued.TrySetResult(); return _release.Task;
        }
        return _real.InvokeAsync(callback, token);
    }
    internal async Task DrainAsync()
    {
        var work = _initial; _initial = null;
        try { if (work != null) await work(); }
        finally { _release.TrySetResult(); }
    }
}
