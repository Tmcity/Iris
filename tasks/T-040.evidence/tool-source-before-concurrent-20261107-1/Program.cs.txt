static string ExtractSource(JsonElement request)
{
    if (!request.TryGetProperty("messages", out JsonElement messages) || messages.GetArrayLength() == 0) return "";
    JsonElement content = messages[messages.GetArrayLength() - 1].GetProperty("content");
    return content.ValueKind == JsonValueKind.String ? content.GetString() ?? ""
        : string.Join("\n", content.EnumerateArray().Where(x => x.TryGetProperty("text", out _)).Select(x => x.GetProperty("text").GetString()));
}
static Task OpenAiDeltaAsync(HttpListenerResponse response, string? text, string? finish, CancellationToken token) =>
    SendAsync(response, "data: " + JsonSerializer.Serialize(new { id = "iris_fixture", @object = "chat.completion.chunk", model = "iris-fixture", choices = new[] { new { index = 0, delta = new { content = text }, finish_reason = finish } } }) + "\n\n", token);
static Task EventAsync(HttpListenerResponse response, string name, object data, CancellationToken token) =>
    SendAsync(response, "event: " + name + "\ndata: " + JsonSerializer.Serialize(data) + "\n\n", token);
static async Task SendAsync(HttpListenerResponse response, string text, CancellationToken token)
{
    byte[] bytes = Encoding.UTF8.GetBytes(text);
    await response.OutputStream.WriteAsync(bytes, token).ConfigureAwait(false);
    await response.OutputStream.FlushAsync(token).ConfigureAwait(false);
}
